param([Parameter(Mandatory=$true)][string]$OutputPath)
$ErrorActionPreference = 'Stop'
if (Test-Path -LiteralPath $OutputPath) { throw 'OUTPUT_ALREADY_EXISTS' }
$identity = [Security.Principal.WindowsIdentity]::GetCurrent()
$principal = New-Object Security.Principal.WindowsPrincipal($identity)
$errorsFound = @()
function Observe([string]$Name, [scriptblock]$Action) {
    try { & $Action } catch { $script:errorsFound += @{part=$Name; error=$_.Exception.Message}; return $null }
}
$report = [ordered]@{
    schema='unified.ls5-pilot-readiness-observation.v1'
    observed_at_utc=[DateTime]::UtcNow.ToString('o')
    computer=$env:COMPUTERNAME
    identity=@{name=$identity.Name; sid=$identity.User.Value; elevated=$principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)}
    os=Observe 'os' { Get-CimInstance Win32_OperatingSystem | Select-Object Caption,Version,BuildNumber,LastBootUpTime,FreePhysicalMemory,FreeVirtualMemory }
    disks=Observe 'disks' { @(Get-CimInstance Win32_LogicalDisk -Filter 'DriveType=3' | Select-Object DeviceID,Size,FreeSpace) }
    services=Observe 'services' { @(Get-CimInstance Win32_Service | Where-Object { $_.Name -match 'unified|consul|postgres|sshd|tailscale|ekrn' } | Select-Object Name,State,StartMode,StartName) }
    listeners=Observe 'listeners' { @(Get-NetTCPConnection -State Listen | Select-Object LocalAddress,LocalPort,OwningProcess) }
    network=Observe 'network' { @(Get-NetIPConfiguration | Select-Object InterfaceAlias,InterfaceIndex,IPv4Address,IPv4DefaultGateway,DNSServer) }
    firewall=Observe 'firewall' { @(Get-NetFirewallProfile | Select-Object Name,Enabled,DefaultInboundAction,DefaultOutboundAction) }
    policies=Observe 'policies' { @(Get-ExecutionPolicy -List | Select-Object Scope,ExecutionPolicy) }
    installations=Observe 'installations' { @(Get-ItemProperty 'HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall\*','HKLM:\Software\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*' -ErrorAction SilentlyContinue | Where-Object { $_.DisplayName -match 'Unified|Consul|Python|Node.js|PostgreSQL|OpenSSH' } | Select-Object DisplayName,DisplayVersion,InstallLocation) }
    candidate_existing_roots=Observe 'roots' { @('C:\Program Files\Unified','C:\ProgramData\Unified','C:\Unified') | ForEach-Object { @{path=$_; exists=(Test-Path -LiteralPath $_)} } }
    backup_admin_channel='NOT_VERIFIED_END_TO_END'
    release_manifest='NOT_ADMITTED'
    backup_restore='NOT_RUN'
    rollback='NOT_RUN'
    install_authorized_by_this_observation=$false
}
$report.errors=$errorsFound
$report.finished_at_utc=[DateTime]::UtcNow.ToString('o')
$json=$report | ConvertTo-Json -Depth 12
[IO.File]::WriteAllText($OutputPath,$json,(New-Object Text.UTF8Encoding($false)))
Write-Output ('READ_ONLY_OBSERVATION_SAVED errors='+$errorsFound.Count)
